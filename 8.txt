CR-1 :- DISPLAY EMPID, EMPLOYEE NAME, MONTHLY_SAL ?

SELECT EMPNO AS EMPID, ENAME AS "EMPLOYEE NAME", SAL AS MONTHLY_SAL
FROM EMP;


CR-2 :- DISPLAY EMPID, EMPLOYEE NAME,  DESIGNATION, ANNUAL_SALARY ?

SELECT EMPNO AS EMPID, ENAME AS "EMPLOYEE NAME", JOB AS DESIGNATION,
              SAL * 12 AS ANNUAL_SAL
              FROM EMP;

CR-3 :- DISPLAY EMPNO, employee_name, MONTHLY_SALARY, COMM, 
PF (SAL*0.12) , HRA (SAL*0.15), DA (SAL*0.10) ,
MONTHLY_GROSS ( MONTHLY_SALARY + COMM+ PF +HRA + DA),
CTC  ( MONTHLY_GROSS*12  ).

SELECT EMPNO, ENAME AS "employee_name", SAL AS MONTHLY_SALARY, 
COMM, SAL * 0.12 AS PF, SAL*0.15 AS HRA, SAL*0.10 AS DA,
SAL + NVL(COMM,0) + SAL * 0.12 +  SAL*0.15 + SAL*0.10 AS MONTHLY_GROSS,
 ( SAL + NVL(COMM,0) + SAL * 0.12 +  SAL*0.15 + SAL*0.10 ) *12 AS CTC
 FROM EMP;

Relation or Comparison operator
===============================
=  equal operator
<>,^=,!= NOT Equal OPERATOR
> ,>= Greater than and greater than equal to
<,<=  Smaller than or smaller than equal
IN  -   To compare Multiple value
NOT IN – NOT IN OPERATOR
IS – Compare value with NULL

=  equal operator
=================
SELECT * FROM EMP WHERE ENAME = 'KING';

<>,^=,!= NOT Equal OPERATOR
=============================
SELECT * FROM EMP WHERE ENAME  <>  'KING';

SELECT * FROM EMP WHERE ENAME  ^=  'KING';

SELECT * FROM EMP WHERE ENAME  !=  'KING';

> ,>= Greater than and greater than equal to
=====================================

SELECT * FROM EMP WHERE SAL > 1000;

SELECT * FROM EMP WHERE HIREDATE > '01-JAN-81';

SELECT * FROM EMP WHERE SAL >= 3000;


<,<=  Smaller than or smaller than equal
=====================================
SELECT * FROM EMP WHERE SAL < 1000;

SELECT * FROM EMP WHERE SAL <= 1000;


SELECT * FROM EMP WHERE HIREDATE < '01-JAN-82';


IN  -   To compare Multiple value
===============================
SELECT * FROM EMP WHERE SAL IN (1000, 2000, 3000);

SELECT * FROM EMP WHERE JOB IN ('MANAGER' , 'ANALYST');

SELECT * FROM EMP WHERE HIREDATE IN ('01-MAY-81' , '03-DEC-81');

NOT IN – NOT IN OPERATOR
===========================

SELECT * FROM EMP WHERE JOB NOT IN ('MANAGER' , 'ANALYST');


IS – Compare value with NULL
============================

SELECT * FROM EMP WHERE COMM IS NULL;


LOGICAL OPERATOR
====================

 A logical condition combines the result of two OR MORE  condition to produce a single result.

. Three logical operators are available in Oracle.

1.AND  2.  OR  3. NOT


1.AND 
========
TRUTH TABLE
TRUE TRUE = TRUE
TRUE FALSE = FALSE
FALSE TRUE  = FALSE
FALSE FALSE  = FALSE

Q:- DISPLAY EMP INFORMATION WHO IS WORKING AS A SALESMAN AND SAL MORE THAN 1300 

SELECT * FROM EMP WHERE JOB = 'SALESMAN' AND SAL > 1300;

SELECT * FROM EMP WHERE JOB = 'SALESMAN' AND SAL > 3000;

SELECT * FROM EMP WHERE JOB = 'SALESMAN' AND SAL > 1300 AND COMM > 300;

2. OR
======
TRUTH TABLE
TRUE TRUE = TRUE
TRUE FALSE = TRUE
